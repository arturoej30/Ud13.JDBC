CREATE TABLE ejemplo (
    id NUMBER PRIMARY KEY,
    nombre VARCHAR2(50)
);

BEGIN
    FOR i IN 1..10 LOOP
        INSERT INTO ejemplo VALUES (i, 'Nombre ' || TO_CHAR(i));
    END LOOP;
    COMMIT;
END;

SELECT * from ejemplo;

CREATE OR REPLACE TRIGGER trg_insert_ejemplo
	BEFORE INSERT ON ejemplo
	FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE('Nuevo registro insertado en la tabla ejemplo.');
END;

INSERT INTO ejemplo VALUES(100, 'Nombre 100');

CREATE OR REPLACE TRIGGER trg_delete_ejemplo
	BEFORE DELETE ON ejemplo
	FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE('El registro ' || :old.id || ' fue eliminado de la tabla ejemplo.');
END;

DELETE FROM ejemplo where id=100;

CREATE OR REPLACE TRIGGER trg_update_nombre_ejemplo
BEFORE UPDATE OF nombre ON ejemplo
FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE('El nombre del registro con ID ' || :OLD.id || ' fue actualizado de "' || :OLD.nombre || '" a "' || :NEW.nombre || '".');
END;

UPDATE ejemplo set nombre='Pepe' WHERE id=100;
UPDATE ejemplo set nombre='María' WHERE id=1;
